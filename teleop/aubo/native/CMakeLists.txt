cmake_minimum_required(VERSION 3.18)
project(aubo_readonly LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
set(AUBO_SDK_ROOT "" CACHE PATH "Existing Linux x64 AUBO SDK directory")
if(NOT EXISTS "${AUBO_SDK_ROOT}/include/aubo_sdk/serviceinterface.h")
  message(FATAL_ERROR "Specify the existing Linux SDK with -DAUBO_SDK_ROOT=...")
endif()

set(AUBO_RELEASE_LIBRARY "${AUBO_SDK_ROOT}/lib/libaubo_sdk.so.2.5.0")
if(NOT EXISTS "${AUBO_RELEASE_LIBRARY}")
  message(FATAL_ERROR "SDK release library not found: ${AUBO_RELEASE_LIBRARY}")
endif()
option(AUBO_BUILD_J6_TRIAL "Build a supervised, one-shot J6 micro-motion trial" OFF)
option(AUBO_BUILD_FOLLOW "Build supervised J6-only continuous follower" OFF)
if(AUBO_BUILD_FOLLOW)
  pybind11_add_module(_aubo_follow_fake ../../../tests/native/fake_follow_binding.cpp)
  target_compile_features(_aubo_follow_fake PRIVATE cxx_std_17)
  target_include_directories(_aubo_follow_fake PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/fake_follow_sdk")
  set_target_properties(_aubo_follow_fake PROPERTIES OUTPUT_NAME "_aubo_follow"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fake-python")
  add_executable(test_follow_profile ../../../tests/native/test_follow_profile.cpp)
  target_compile_features(test_follow_profile PRIVATE cxx_std_17)
  target_include_directories(test_follow_profile PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_executable(test_passthrough ../../../tests/native/test_passthrough.cpp)
  target_compile_features(test_passthrough PRIVATE cxx_std_17)
  target_include_directories(test_passthrough PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  pybind11_add_module(_aubo_follow follow.cpp)
  target_compile_features(_aubo_follow PRIVATE cxx_std_17)
  target_include_directories(_aubo_follow PRIVATE "${AUBO_SDK_ROOT}/include")
  target_link_libraries(_aubo_follow PRIVATE "${AUBO_RELEASE_LIBRARY}")
  set_target_properties(_aubo_follow PROPERTIES BUILD_RPATH "${AUBO_SDK_ROOT}/lib"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
endif()
if(AUBO_BUILD_J6_TRIAL)
  pybind11_add_module(_aubo_j6_trial j6_trial.cpp)
  target_compile_features(_aubo_j6_trial PRIVATE cxx_std_17)
  target_include_directories(_aubo_j6_trial PRIVATE "${AUBO_SDK_ROOT}/include")
  target_link_libraries(_aubo_j6_trial PRIVATE "${AUBO_RELEASE_LIBRARY}")
  set_target_properties(_aubo_j6_trial PROPERTIES
    BUILD_RPATH "${AUBO_SDK_ROOT}/lib"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
endif()
pybind11_add_module(_aubo_readonly bindings.cpp)
target_compile_features(_aubo_readonly PRIVATE cxx_std_17)
target_include_directories(_aubo_readonly PRIVATE "${AUBO_SDK_ROOT}/include")
target_link_libraries(_aubo_readonly PRIVATE "${AUBO_RELEASE_LIBRARY}")
set_target_properties(_aubo_readonly PROPERTIES
  BUILD_RPATH "${AUBO_SDK_ROOT}/lib"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")

option(AUBO_BUILD_STOP_PROBE "Build a separate stationary stop diagnostic; no movement API" OFF)
if(AUBO_BUILD_STOP_PROBE)
  pybind11_add_module(_aubo_stop_probe stop_probe.cpp)
  target_compile_features(_aubo_stop_probe PRIVATE cxx_std_17)
  target_include_directories(_aubo_stop_probe PRIVATE "${AUBO_SDK_ROOT}/include")
  target_link_libraries(_aubo_stop_probe PRIVATE "${AUBO_RELEASE_LIBRARY}")
  set_target_properties(_aubo_stop_probe PROPERTIES
    BUILD_RPATH "${AUBO_SDK_ROOT}/lib"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
endif()
